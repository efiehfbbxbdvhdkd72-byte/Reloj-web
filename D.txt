<script>
function updateClock() {
const now = new Date();
let hours = now.getHours();
const minutes = now.getMinutes();
const seconds = now.getSeconds();

// Formato de 12 horas con AM/PM
const ampm = hours >= 12 ? 'PM' : 'AM';
hours = hours % 12;
hours = hours ? hours : 12; // La hora '0' se convierte en '12'

const formattedHours = String(hours).padStart(2, '0');
const formattedMinutes = String(minutes).padStart(2, '0');
const formattedSeconds = String(seconds).padStart(2, '0');

// Mostrar hora actual exacta del dispositivo
document.getElementById('time-display').textContent = ${formattedHours}:${formattedMinutes}:${formattedSeconds}`;
document.getElementById('ampm-display').textContent = ampm;

// Cuenta regresiva para el siguiente cambio de minuto
const secondsLeft = 60 - seconds;
document.getElementById('seconds-remaining').textContent = ``${secondsLeft}s`;

// Ancho de la barra de progreso
const progressPercent = (secondsLeft / 60) * 100;
document.getElementById('progress-bar').style.width = ${progressPercent}%;
}

// Ejecutar inmediatamente y actualizar cada segundo
updateClock();
setInterval(updateClock, 1000);
</script>
